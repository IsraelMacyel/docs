# Account Information

> Get information about your account.

<ParamField header="Authorization" type="string" placeholder="API Key" required>
  The API key for your account. You can find this in your [account settings](https://virtuscloud.app/account/security).
</ParamField>

### Response

<ResponseField name="status" type="string">
  Indicates whether the call was successful.. `success` if successful, `error` if not.
</ResponseField>

<ResponseField name="response" type="object">
  The response content.

  <Expandable title="Toggle object">
    <ResponseField name="user" type="object">
      <Expandable title="Toggle object">
        <ResponseField name="id" type="string">
          The user's ID.
        </ResponseField>

        <ResponseField name="name" type="string">
          The user's name.
        </ResponseField>

        <ResponseField name="email" type="string">
          The user's email.
        </ResponseField>

        <ResponseField name="locale" type="string">
          The user's locale.
        </ResponseField>

        <ResponseField name="plan" type="object">
          <Expandable title="Toggle object">
            <ResponseField name="name" type="string">
              The user's plan name.
            </ResponseField>

            <ResponseField name="memory" type="object">
              The user's plan memory details.

              <Expandable title="Toggle object">
                <ResponseField name="limit" type="number">
                  The user's plan memory limit in MB.
                </ResponseField>

                <ResponseField name="available" type="number">
                  The user's plan available memory in MB.
                </ResponseField>

                <ResponseField name="used" type="number">
                  The user's plan used memory in MB.
                </ResponseField>
              </Expandable>
            </ResponseField>

            <ResponseField name="duration" type="number">
              The user's plan duration in milliseconds.
            </ResponseField>
          </Expandable>
        </ResponseField>

        <ResponseField name="created_at" type="date">
          The user's account creation date.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="applications" type="array">
      <Expandable title="Toggle object">
        <ResponseField name="name" type="string">
          The application's name.
        </ResponseField>

        <ResponseField name="id" type="string">
          The application's ID.
        </ResponseField>

        <ResponseField name="desc" type="string">
          The application's description.
        </ResponseField>

        <ResponseField name="ram" type="number">
          The application's RAM usage.
        </ResponseField>

        <ResponseField name="lang" type="string">
          The application's language.
        </ResponseField>

        <ResponseField name="domain" type="string">
          The application's domain.
        </ResponseField>

        <ResponseField name="custom" type="string">
          The application's custom domain.
        </ResponseField>

        <ResponseField name="cluster" type="string">
          The application's cluster.
        </ResponseField>

        <ResponseField name="created_at" type="date">
          The application's creation date.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="databases" type="array">
      <Expandable title="Toggle object">
        <ResponseField name="id" type="string">
          The database's ID.
        </ResponseField>

        <ResponseField name="name" type="string">
          The database's name.
        </ResponseField>

        <ResponseField name="ram" type="number">
          The database's RAM usage.
        </ResponseField>

        <ResponseField name="type" type="string">
          The database's type.
        </ResponseField>

        <ResponseField name="cluster" type="string">
          The database's cluster.
        </ResponseField>

        <ResponseField name="created_at" type="date">
          The database's creation date.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
